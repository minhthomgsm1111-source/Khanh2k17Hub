-- Universal GUI Protection Function (Hỗ trợ tất cả Client)
local function GetSafeParent()
	local success, result = pcall(function()
		if gethui then
			return gethui()
		elseif syn and syn.protect_gui then
			local sg = Instance.new("ScreenGui")
			syn.protect_gui(sg)
			return sg.Parent
		elseif CoreGui then
			return CoreGui
		else
			return LocalPlayer:WaitForChild("PlayerGui")
		end
	end)
	
	if success and result then
		return result
	else
		return LocalPlayer:WaitForChild("PlayerGui")
	end
end

-- Logic Kéo Vuốt (Drag) đa nền tảng
local dragging, dragInput, dragStart, startPos
local function updateInput(input)
	local delta = input.Position - dragStart
	OpenBtn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
end

OpenBtn.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = OpenBtn.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				dragging = false
			end
		end)
	end
end)

OpenBtn.InputChanged:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
		if dragging then
			updateInput(input)
		end
	end
end)

-- Hàm Hop Server (Chính xác số người)
local function HopServer()
	local PlaceId = game.PlaceId
	local JobId = game.JobId
	
	HopBtn.Text = "⏳ Đang tìm server " .. ExactHopPlayers .. " người..."
	
	local success, result = pcall(function()
		return HttpService:JSONDecode(
			game:HttpGet("https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?sortOrder=Asc&limit=100")
		)
	end)

	if success and result and result.data then
		for _, server in ipairs(result.data) do
			if server.playing and server.playing == ExactHopPlayers and server.id ~= JobId then
				TeleportService:TeleportToPlaceInstance(PlaceId, server.id, LocalPlayer)
				return
			end
		end
	end
	
	HopBtn.Text = "❌ Không thấy SV " .. ExactHopPlayers .. " người!"
	task.wait(2)
	HopBtn.Text = "🌐 Hop"
end

HopBtn.MouseButton1Click:Connect(HopServer)

-- Target Finder
local function GetTargetPlayer()
	if TargetName == "" or TargetName == " " then return nil end
	local lowerInput = string.lower(TargetName)
	for _, player in pairs(Players:GetPlayers()) do
		if player ~= LocalPlayer then
			if string.lower(player.Name) == lowerInput or string.lower(player.DisplayName) == lowerInput then
				return player
			end
		end
	end
	return nil
end

-- Render Loop cho Aim

	if AimTargetEnabled and target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
		Camera.CFrame = CFrame.new(Camera.CFrame.Position, target.Character.HumanoidRootPart.Position)
	end
end)
