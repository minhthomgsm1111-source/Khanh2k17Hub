-- Universal GUI Protection Function
local function GetSafeParent()
	local success, result = pcall(function()
		if gethui then
			return gethui()
		elseif syn and syn.protect_gui then
			local sg = Instance.new("ScreenGui")
			syn.protect_gui(sg)
			return sg.Parent
		elseif CoreGui then
			return CoreGui
		else
			return LocalPlayer:WaitForChild("PlayerGui")
		end
	end)
	return (success and result) and result or LocalPlayer:WaitForChild("PlayerGui")
end

local btnDragging = false

	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		btnDragging = true
		btnDragMoved = false
		btnDragStart = input.Position
		btnStartPos = OpenBtn.Position
	end

OpenBtn.InputChanged:Connect(function(input)
	if (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) and btnDragging then
		local delta = input.Position - btnDragStart
		if delta.Magnitude > 8 then
			btnDragMoved = true
			OpenBtn.Position = UDim2.new(btnStartPos.X.Scale, btnStartPos.X.Offset + delta.X, btnStartPos.Y.Scale, btnStartPos.Y.Offset + delta.Y)
		end
	end
end)

OpenBtn.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		btnDragging = false
	end
end)

local function UpdatePlayerList()
	for _, child in pairs(DropdownScroll:GetChildren()) do
		if child:IsA("TextButton") then child:Destroy() end
	end
	
	local count = 0
	for _, plr in pairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer then
			count = count + 1
			local Btn = Instance.new("TextButton")
			Btn.Size = UDim2.new(1, 0, 0, 25)
			Btn.BackgroundColor3 = Color3.fromRGB(32, 32, 44)
			Btn.Text = plr.DisplayName .. " (@" .. plr.Name .. ")"
			Btn.TextColor3 = Color3.fromRGB(220, 220, 220)
			Btn.TextSize = 10
			Btn.Font = Enum.Font.Gotham
			Btn.Parent = DropdownScroll
			
			Btn.MouseButton1Click:Connect(function()
				TargetPlayer = plr
				DropdownBtn.Text = "👤 Mục Tiêu: " .. plr.DisplayName
				DropdownFrame.Visible = false
			end)
		end
	end
	DropdownScroll.CanvasSize = UDim2.new(0, 0, 0, count * 25)
end

DropdownBtn.MouseButton1Click:Connect(function()
	DropdownFrame.Visible = not DropdownFrame.Visible
	if DropdownFrame.Visible then UpdatePlayerList() end
end)

local function HopServer()
	local PlaceId = game.PlaceId
	local JobId = game.JobId
	HopBtn.Text = "⏳ Đang quét SV " .. ExactHopPlayers .. " người..."
	
	local cursor = ""
	local foundServer = nil
	
	for i = 1, 5 do
		local url = "https://games.roblox.com/v1/games/" .. PlaceId .. "/servers/Public?sortOrder=Asc&limit=100"
		if cursor ~= "" then url = url .. "&cursor=" .. cursor end
		
		local success, result = pcall(function()
			return HttpService:JSONDecode(game:HttpGet(url))
		end)
		
		if success and result and result.data then
			for _, server in ipairs(result.data) do
				if server.playing and server.playing == ExactHopPlayers and server.id ~= JobId then
					foundServer = server.id
					break
				end
			end
			if foundServer or not result.nextPageCursor then break end
			cursor = result.nextPageCursor
		else
			break
		end
	end
	
	if foundServer then
		HopBtn.Text = "🚀 Đang chuyển server..."
		TeleportService:TeleportToPlaceInstance(PlaceId, foundServer, LocalPlayer)
	else
		HopBtn.Text = "❌ Không thấy SV " .. ExactHopPlayers .. " người!"
		task.wait(2)
		HopBtn.Text = "🌐 Tìm & Chuyển Server"
	end
end
HopBtn.MouseButton1Click:Connect(HopServer)

-- --- LOGIC VUỐT TRÁI/PHẢI CHUYỂN TAB ---
local currentTab = 1
local touchStartPos = nil
local swipeDragging = false

MainFrame.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		touchStartPos = input.Position
		swipeDragging = true
	end
end)

MainFrame.InputEnded:Connect(function(input)
	if (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) and swipeDragging and touchStartPos then
		swipeDragging = false
		local deltaX = input.Position.X - touchStartPos.X
		local deltaY = input.Position.Y - touchStartPos.Y
		
		-- Chỉ nhận vuốt ngang nếu khoảng cách vuốt > 40px và nghiêng hơn chiều dọc
		if math.abs(deltaX) > 40 and math.abs(deltaX) > math.abs(deltaY) then
			local tweenInfo = TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
			if deltaX < 0 and currentTab == 1 then
				-- Vuốt sang Trái -> Sang Tab 2
				currentTab = 2
				TweenService:Create(TabContainer, tweenInfo, {Position = UDim2.new(-1, 0, 0, 40)}):Play()
				PageIndicator.Text = "○ ●  (Trang 2 / 2 - Vuốt về)"
			elseif deltaX > 0 and currentTab == 2 then
				-- Vuốt sang Phải -> Về Tab 1
				currentTab = 1
				TweenService:Create(TabContainer, tweenInfo, {Position = UDim2.new(0, 0, 0, 40)}):Play()
				PageIndicator.Text = "● ○  (Trang 1 / 2 - Vuốt sang)"
			end
		end
	end
end)

-- CLICK NÚT "K" ĐỂ MỞ/ĐÓNG MENU
local menuOpen = true
OpenBtn.MouseButton1Click:Connect(function()
	if btnDragMoved then return end
	
	menuOpen = not menuOpen
	local tweenInfo = TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
	
	if menuOpen then
		MainFrame.Visible = true
		TweenService:Create(MainScale, tweenInfo, {Scale = 1}):Play()
	else
		local tweenScale = TweenService:Create(MainScale, tweenInfo, {Scale = 0.1})
		tweenScale:Play()
		tweenScale.Completed:Connect(function()
			if not menuOpen then MainFrame.Visible = false end
		end)
	end
end)

-- VÒNG LẶP XỬ LÝ CHÍNH
RunService.RenderStepped:Connect(function(dt)
	local myChar = LocalPlayer.Character
	if not myChar then return end
	local myHRP = myChar:FindFirstChild("HumanoidRootPart")
	local myHum = myChar:FindFirstChildOfClass("Humanoid")
	if not myHRP or not myHum or myHum.Health <= 0 then return end

	-- Reset Hitbox nếu tắt hoặc chưa chọn target
	if not HitboxEnabled or not TargetPlayer then
		for _, plr in pairs(Players:GetPlayers()) do
			if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") then
				local hrp = plr.Character.HumanoidRootPart
				if hrp.Size ~= Vector3.new(2, 2, 1) then
					hrp.Size = Vector3.new(2, 2, 1)
					hrp.Transparency = 1
				end
			end
		end
	end

	-- Kiểm tra Mục tiêu
	if not TargetPlayer or not TargetPlayer.Character then return end
	local targetChar = TargetPlayer.Character
	local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
	local targetHum = targetChar:FindFirstChildOfClass("Humanoid")
	
	if not targetHRP or not targetHum or targetHum.Health <= 0 then return end
	
	-- 1. FLY SAU LƯNG (3.5 studs)
	if FlyBehindEnabled then
		for _, part in pairs(myChar:GetChildren()) do
			if part:IsA("BasePart") then part.CanCollide = false end
		end
		
		local targetCFrame = targetHRP.CFrame
		local behindPosition = targetCFrame.Position - (targetCFrame.LookVector * DISTANCE_BEHIND)
		
		local currentPos = myHRP.Position
		local direction = (behindPosition - currentPos)
		local distance = direction.Magnitude
		
		if distance > 0.7 then
			local moveStep = math.min(distance, FLY_SPEED * dt)
			myHRP.CFrame = CFrame.new(currentPos + direction.Unit * moveStep, targetHRP.Position)
		else
			myHRP.CFrame = CFrame.new(behindPosition, targetHRP.Position)
		end
		myHRP.AssemblyLinearVelocity = Vector3.zero
	end
	
	-- 2. HITBOX 10M
	if HitboxEnabled then
		targetHRP.Size = Vector3.new(10, 10, 10)
		targetHRP.Transparency = 0.6
		targetHRP.BrickColor = BrickColor.new("Really red")
		targetHRP.Material = Enum.Material.Neon
		targetHRP.CanCollide = false
	end
	
	-- 3. AIM CAMERA
	if AimTargetEnabled then
		Camera.CFrame = CFrame.new(Camera.CFrame.Position, targetHRP.Position)
	end
end)
